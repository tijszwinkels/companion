cmake_minimum_required(VERSION 3.21)
project(CompanionWindow LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 WebEngineQuick Network)
option(COMPANION_SIDEBAR "Build Wayland sidebar/dock support" ON)
include(CTest)
if(COMPANION_SIDEBAR)
    find_package(LayerShellQt 6.6 REQUIRED)
    find_package(Qt6 REQUIRED COMPONENTS WaylandClient GuiPrivate)
endif()

add_library(companionwindow STATIC connection.cpp host.cpp layer-focus.cpp rendering.hpp)
target_link_libraries(companionwindow PUBLIC Qt6::Quick Qt6::Network)
if(COMPANION_SIDEBAR)
    target_link_libraries(companionwindow PUBLIC LayerShellQt::Interface PRIVATE Qt6::WaylandClient Qt6::GuiPrivate)
    target_compile_definitions(companionwindow PRIVATE COMPANION_LAYER_SHELL)
    qt_generate_wayland_protocol_client_sources(companionwindow FILES "${CMAKE_CURRENT_SOURCE_DIR}/protocols/hyprland-focus-grab-v1.xml")
endif()
add_executable(companion-ui main.cpp)
target_link_libraries(companion-ui PRIVATE companionwindow Qt6::WebEngineQuick)
target_compile_definitions(companion-ui PRIVATE
    COMPANION_UI_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/../../integrations/window/Main.qml")

add_library(companionwidgets MODULE ../widget-renderer/profile.cpp ../widget-renderer/plugin.cpp)
target_link_libraries(companionwidgets PRIVATE Qt6::WebEngineQuick Qt6::Network)
set_target_properties(companionwidgets PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qml/Companion/Widgets")
configure_file(../widget-renderer/qmldir "${CMAKE_BINARY_DIR}/qml/Companion/Widgets/qmldir" COPYONLY)

if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS QuickTest)
    add_executable(companion-window-tests test-main.cpp)
    target_link_libraries(companion-window-tests PRIVATE companionwindow Qt6::WebEngineQuick Qt6::QuickTest)
endif()

install(TARGETS companion-ui RUNTIME DESTINATION bin)
install(TARGETS companionwidgets LIBRARY DESTINATION lib/companion/qml/Companion/Widgets)
install(FILES ../widget-renderer/qmldir DESTINATION lib/companion/qml/Companion/Widgets)
install(DIRECTORY ../../integrations/window ../../integrations/panel DESTINATION share/companion/integrations)
install(FILES companion.desktop DESTINATION share/applications)
install(FILES protocols/hyprland-focus-grab-v1.xml DESTINATION share/licenses/companion-ui)
